# BLOCK_ENTITY_SHADOWS: activado -> false   idem con cofres, carteles y maquinas; con 270 mods son cientos por chunk
BLOCK_ENTITY_SHADOWS=false
# COLORED_SHADOWS: activado -> false   triplica las lecturas por pixel sombreado
COLORED_SHADOWS=false
# LIGHT_SHAFT: activado -> false   raymarch de 10 muestras siempre activo
LIGHT_SHAFT=false
# REALISTIC_CLOUDS: 1 -> 0   nubes volumetricas siempre activas, raymarch de 10 muestras
#   legales: 0 1 2
REALISTIC_CLOUDS=0
# SHADOWS_ENTITIES: activado -> false   shaders.properties lo traduce a shadowEntities=false: Oculus deja de rasterizar entidades en el pase de sombra
SHADOWS_ENTITIES=false
# SSS: activado -> false   subsurface scattering con muestreo del shadow map en bucle
SSS=false
# shadowDistance: 512.0 -> 128.0   tope 128 bloques (tu render distance son 192)
#   legales: 32.0 64.0 96.0 128.0 160.0 192.0 224.0 256.0 288.0 320.0 352.0 384.0 416.0 448.0 480.0 512.0 544.0 576.0 608.0 640.0 672.0 704.0 736.0 768.0 800.0 832.0 864.0 896.0 928.0 960.0 992.0 1024.0
shadowDistance=128.0
# shadowMapResolution: 3072 -> 1024   tope 1024 en una GPU de 4 GB
#   legales: 512 1024 1536 2048 2560 3072 3584 4096 4608 5120 5632 6144 6656 7168 7680 8192
shadowMapResolution=1024
